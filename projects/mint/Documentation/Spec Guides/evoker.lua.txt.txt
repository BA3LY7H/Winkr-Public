	end
	if specId == 256 then -- Discipline Priest

	if not u or not u.exists or not u.visible then
		return 0
	end

	if u.specId == 71 and (u.cd(262161) > 35 or u.cd(167105) > 35 and not u.disarmed) then
		if u.target and u.target.exists and u.meleeRangeOf(u.target) and u.target.debuff(307865) then
			return u.target.debuffUptime(307865)
		end
		if u.target and u.target.exists and u.target.debuff(208086, u) then
			return u.target.debuffUptime(208086)
		end
	end

	if u.class2 == "MAGE" then
		if u.buff(190319) then
			return u.buffUptime(190319)
		end
		if u.buff(12472) then
			return u.buffUptime(12472)
		end
		if u.buff(365362) and u.specId == 62 then
			return u.buffUptime(365362)
		end
		return 0
	end

	if u.specId == 72 and not u.disarmed then
		if u.buffRemains(18499) >= 4 then
			return u.buffUptime(18499)
		end
		if u.target and u.target.exists then
			local tar = u.target
			if tar.debuff(388933, u) and tar.debuffStacks(388933) >= 8 then
				return tar.debuffUptime(388933)
			end
		end
		return 0
	end

	if u.specId == 260 and not u.disarmed then
		if u.debuff(343142) then
			return u.debuffUptime(343142)
		end
		if u.buff(13750) then
			return u.buffUptime(13750)
		end
	end

	if u.specId == 252 then
		if u.buff(315443) then
			return u.buffUptime(315443)
		end
	end

	if u.specId == 259 and not u.disarmed then
		if u.target and u.target.exists then
			if u.target.debuff(385408) then
				return u.target.debuffUptime(385408)
			end
			if u.target.debuff(360194) then
				return u.target.debuffUptime(360194)
			end
		end
	end

	if u.class2 == "ROGUE" and not u.disarmed then
		if u.target and u.target.exists and u.meleeRangeOf(u.target) and u.target.debuff(384631) then
			return u.target.debuffUptime(384631)
		end
		if u.buff(121471) then
			return u.buffUptime(121471)
		end
	end

	if u.specId == 577 then
		local tar = u.target
		if tar.exists and tar.debuff(320338) and tar.losOf(u) then
			return tar.debuffUptime(320338)
		end
	end

	if u.specId == 266 then
		if u.buff(267217) then
			return u.buffUptime(267217)
		end
		local cd = GetSpellBaseCooldown(265187)
		cd = cd * 0.001
		local cdCur = u.cd(265187)
		if cdCur > 0 and cdCur >= cd - 15 then
			return math.max(0, cd - cdCur)
		end
		local cdGrimmoire = u.cd(111898)
		if cdGrimmoire > 108 and cdGrimmoire < 120 then
			return 120 - cdGrimmoire
		end
	end

	return u.burstingSince or 0
end)

local function getOffTarget()
	if not awful.arena then
		return nil
	end

	local currentTarget = awful.target
	if not currentTarget or not currentTarget.isPlayer or not currentTarget.exists then
		return nil
	end

	local arenaBracket = GetNumGroupMembers()

	if arenaBracket == 2 then
		-- 2v2: return the enemy that's not your current target
		for _, enemy in pairs(awful.enemies) do
			if enemy.isPlayer and not enemy.isUnit(currentTarget) then
				return enemy
			end
		end
	elseif arenaBracket == 3 then
		-- 3v3: return the enemy that's not your current target and not the healer
		for _, enemy in pairs(awful.enemies) do
			if enemy.isPlayer and not enemy.isUnit(currentTarget) and not enemy.healer then
				return enemy
			end
		end
	end

	return nil
end

local function evokerStomp()
	awful.enemyTotems.loop(function(totem, uptime)
		if not totem.exists or not totem.enemy or uptime < awful.delays.pvpDelay.now or totem.distance > 40 then
			return
		end

		local entry = awful.Lists.pvpStomps[totem.id]
		if not entry then
			return
		end

		if entry.fragile then
			if dev.azureStrike:Castable(totem) then
				return dev.azureStrike:Cast(totem)
			end
		elseif entry.highPrio then
			if player.buff(buffs.burnOut) and totem.hp > dev.azureStrike.effect and dev.livingFlame:Castable(totem) then
				return dev.livingFlame:Cast(totem)
			elseif dev.azureStrike:Castable(totem) then
				return dev.azureStrike:Cast(totem)
			end
		end
	end)
end

	:Callback(function(spell)
		if player.buff(buffs.tipTheScales) then
			return
		end

			return
		end
		if not spell:Castable(target) then

		end

		if awful.burst and dev.dragonRage.cd < 10 then
			return
		end

		local enemiesInArc = #awful.allEnemies.arc(
			player.x,
			player.y,
			player.z,
			25, -- range in yards
			30, -- arc angle in degrees
			player.rotation,
			function(unit)
				return not unit.bcc and not unit.immuneMagic and unit.los
			end
		)

		if enemiesInArc > 0 then
			return spell:Cast(target)
		end
	end)

		if player.empowerStage == 1 then
			return spell:Release()
		end
	end)

local buffblacklist = {}
awful:OnProjectEvent(function(info, source, dest)
	local subEvent, spellName = info[2], info[13]
	if subEvent == "SPELL_AURA_REMOVED" then
		if not dest then
			dest = awful.Object:FromGUID(info.destGUID)
		end
		if dest.friend and spellName == dev.blessingOfTheBronze.name then
			buffblacklist[dest.guid] = awful.time + awful.delays.xxl.now
		end
	end
end)

dev.blessingOfTheBronze:Callback(function(spell)
	if spell.used(awful.delays.xxxl.now) then
		return
	end
	awful.fullGroup.loop(function(unit)
		if
			buffblacklist[unit.guid]
			or not spell:Castable(unit)
			or unit.distance > 40
			or unit.buffRemains(spell.name) > (awful.delays.xxxl.now * 5)
		then
			return
		end
		spell:Cast(player)
	end)
end)

dev.sourceOfMagic:Callback(function(spell)
	if not awful.pvp then
		return
	end
	if not awful.healer or not awful.healer.exists then
		return
	end
	if awful.healer.buff(spell.name) then
		return
	end
	local prepTime = inPrep()
	if prepTime < 30 or prepTime > 45 then
		return
	end
	return spell:Cast(awful.healer)
end)

dev.verdantEmbrace:Callback(function(spell)
	if player.buff(buffs.hover) then
		return
	end
	if player.buff(buffs.obsidianScales) then
		return
	end
	if awful.enemies.around(player, 5) == 0 then
		return
	end
	if player.attackers() < 1 then
		return
	end

	local bestTarget = nil
	awful.fullGroup.loop(function(unit)
		if not unit.exists or unit.dead then
			return
		end
		if not spell:Castable(unit) then
			return
		end
		if unit.distance > 25 then
			return
		end

		local enemiesNearUnit = awful.enemies.around(unit, 8)
		if enemiesNearUnit == 0 then
			if not bestTarget or unit.hp < bestTarget.hp then
				bestTarget = unit
			end
		end
	end)

	if bestTarget then
		return spell:Cast(bestTarget)
	end
end)

	if dontStop[player.chastId] and dontStop[player.chastId]() then
		awful.alert("Not kicking - dontStop active", player.chastId)
		return
	end
	awful.enemies.loop(function(unit)
		if unit.dead or unit.friend or not unit.chasting or unit.cannotBeInterrupted then
			return
		end
		if player.chasting and player.chastRemains < unit.chastRemains + awful.buffer then
			return
		end
		local ct = unit.castTarget
		if ct and ct.friend and ct.buff(8178) then
			return
		end
		if
			player.chasting
			and routine.InterruptCCList[player.chastId]
			and player.castTarget.isUnit(unit)
			and routine.InterruptCCList[player.chastId](unit)
			and player.chastRemains < unit.chastRemains
		then
			return
		end
		if unit.chasting then
			if unit.chastElapsed > routine.InterruptDelay.now and unit.chastRemains > awful.buffer then
				if
					unit.chastRemains < awful.buffer + awful.latency + 0.03 + routine.InterruptFinishDelay.now
					--or not unit.predictLos(unit.chastRemains - awful.buffer + awful.latency + 0.03)
				then
					if
						routine.InterruptSpecialList[unit.chastId]
						or (routine.InterruptCCList[unit.chastId] and routine.InterruptCCList[unit.chastId](ct))
					then
						return not player.chasting and spell:Cast(unit)
					end
					if
						ct
						and ct.friend
						and ct.hp
						and (ct.hp < 35 or (unit.cds or routine.DangerCasts[unit.chastId]) and ct.hp < 60)
					then
						if unit.healer then
							return
						end
						return not player.chasting and spell:Cast(unit)
					end
					if ct and ct.enemy and ct.hp and ct.hp < 60 then
						if awful.enemyHealer and awful.enemyHealer.exists and not awful.enemyHealer.dead and not unit.healer then
							return
						end
						if routine.InterruptChanneledHealsList[unit.chastId] then
							return spell:Cast(unit)
						end
						return spell:Cast(unit)
					end
				end
			end
		end
	end)
end)

routine.cancelCast = {}

awful:OnProjectEvent(function(info, source, dest)
	local subEvent, spellName = info[2], info[13]
	local time = awful.time

	if subEvent == "SPELL_CAST_SUCCESS" then
		if not source then
			source = awful.Object:FromGUID(info.sourceGUID)
		end
		if source then
			if source.enemy then
				if spellName == dev.shadowWordDeath.name then
					if player.casting == dev.sleepWalk.name and player.castTarget.isUnit(source) then
						awful.alert(player.castTarget.name .. " on hex", dev.shadowWordDeath.id)
						routine.cancelCast[source] = {
							cancel = time + awful.delays.tiny.now,
							dmg = time + 1,
							spellid = player.castId,
							finish = time + player.chastRemains,
						}
					end
				end
			end
		end
	elseif subEvent == "SPELL_AURA_APPLIED" then
		if not dest then
			dest = awful.Object:FromGUID(info.destGUID)
		end
		if dest and dest.enemy and info[12] == 23920 then -- Spell Reflection buff ID
			if player.casting == dev.sleepWalk.name and player.castTarget.isUnit(dest) then
				awful.alert("Canceling Hex - " .. dest.name .. " got Spell Reflection", dev.sleepWalk.id)
				routine.cancelCast[dest] = {
					cancel = time + awful.delays.tiny.now,
					dmg = time + 1,
					spellid = player.castId,
					finish = time + player.chastRemains,
				}
			end
		end
	end
end)

awful:OnProjectUpdate(function()
	for enemy, info in pairs(routine.cancelCast) do
		if (player.castId == info.spellid) and (awful.time > info.cancel) and (awful.time < info.finish) then
			if awful.time + player.chastRemains > info.dmg then
				return
			end
			SpellStopCasting()
			SpellStopCasting()
			return
		end
	end
end)

local function safeToSleepWalk(target)
	local ct = dev.sleepWalk.castTime
	local magicImmune = target.immunities and target.immunities.magic and target.immunities.magic.effects
	if magicImmune and magicImmune.remains > ct then
		return false
	end
	if target.buffRemains(23920) >= ct then -- Spell Reflection
		return false
	end
	if target.buffRemains(335255) >= ct then -- Divine Shield
		return false
	end
	if target.buffRemains(210256) >= ct then -- Divine Shield
		return false
	end
	if target.buffRemains(212704) >= ct then -- Divine Shield
		return false
	end
	if target.class2 == "PALADIN" then
		if target.buffRemains(6940) >= ct then -- Divine Shield
			return false
		end
	end
	if target.buffRemains(20711) >= ct then -- Priest Angle
		return false
	end
	return true
end

dev.sleepWalk:Callback("auto", function(spell)
	if player.buff(buffs.dragonRage) then
		return
	end

	-- if not awful.arena then
	--     return
	-- end

	if player.casting == spell.name then
		return
	end

	local sleepy = awful.enemies.find(function(enemy)
		return enemy.enemy and enemy.debuffRemains__insta(spell.name, player) > spell.castTime + awful.buffer
	end)

	if sleepy and sleepy.enemy then
		return
	end

	local tremor = awful.enemyTotems.find(function(totem)
		return totem.id == 5913 and totem.distance <= 32
	end)

	if tremor then
		return
	end

	local function conditionHeal(target)
		if not target.exists then
			return false
		end
		if awful.target.exists and target.isUnit(awful.target) then
			return false
		end
		if
			awful.dps
			and awful.dps.exists
			and awful.dps.target.isUnit(target)
			and not awful.dps.healer
			and awful.dps.ccRemains <= 0
		then
			return false
		end

		local condDR = target.dr.disorient.full
			or (target.dr.disorient.half and target.dr.disorient.remains > 15)
			or (target.dr.disorient.remains > 0 and target.dr.disorient.remains < spell.castTime)
		if not condDR then
			return false
		end

		if not spell:Castable(target) then
			return false
		end

		if target.ccRemains > spell.castTime + awful.GCD.total + awful.buffer then
			return false
		end

		if not safeToSleepWalk(target) then
			return false
		end

		if target.importantLockoutRemains > spell.castTime + player.gcdRemains + awful.buffer then
			return false
		end

		return true
	end

	if awful.enemyHealer.exists then
		if conditionHeal(awful.enemyHealer) then
			return spell:Cast(awful.enemyHealer) and awful.alert("Sleep Walk on Healer", spell.id)
		end
	end

	local sleepWalkReason = ""
	local function conditionDPS(target)
		if not target.exists or not target.visible then
			return false
		end
		if not spell:Castable(target) then
			return false
		end

		-- Never Sleep Walk your current target
		if awful.target.exists and target.isUnit(awful.target) then
			return false
		end

		local targetTarget = target.target
		local targetCast = target.castTarget

		if not targetTarget.exists and (not targetCast or not targetCast.exists) and not target.dummy then
			return false
		end

		if target.ranged and not spell:Castable(target) then
			return false
		end

		local condDR = target.dr.disorient.full
			or (target.dr.disorient.half and target.dr.disorient.remains > 10)
			or (target.dr.disorient.remains > 0 and target.dr.disorient.remains < spell.castTime)
		if not condDR then
			return false
		end

		if not safeToSleepWalk(target) then
			return false
		end

		if target.castIdEx == 116858 and target.castTimeLeft > spell.castTime then -- Chaos Bolt
			sleepWalkReason = "Chaos Bolt"
			return true
		end

		if target.class2 == "ROGUE" then
			if target.buffRemains(185313) > 0 or targetTarget.debuffRemains(408, target) > 0 then -- Shadow Dance, Kidney Shot
				sleepWalkReason = "Rogue Burst"
				return true
			end
		end

		if target.advancedBurstSince > 0.7 then
			sleepWalkReason = "Burst"
			return true
		end

		if target.melee and target.targetsMe and target.distance <= 10 and not target.isUnit(awful.target) then
			sleepWalkReason = "Melee Burst"
			return true
		end

		if awful.arena then
			local lowest = awful.fullGroup.lowest()
			if
				lowest
				and lowest.exists
				and lowest.hp <= 40
				and (targetTarget.isUnit(lowest) or targetCast and lowest.isUnit(targetCast))
			then
				sleepWalkReason = "Friend in danger"
				return true
			end

			local off = getOffTarget()
			if awful.healer.exists and awful.healer.ccRemains > 4 then
				if
					off
					and off.exists
					and target.isUnit(off)
					and target.ccRemains <= awful.GCD.total
					and spell:Castable(off)
				then
					sleepWalkReason = "Melee on you"
					return true
				end
			end
		end

		return false
	end

	local enemy = awful.enemies.find(function(enemy)
		return spell:Castable(enemy)
			and conditionDPS(enemy)
			and enemy.ccRemains <= awful.buffer * 3
			and safeToSleepWalk(enemy)
	end)
	if enemy then
		return spell:Cast(enemy) and awful.alert("Sleep Walk: " .. sleepWalkReason, spell.id)
	end
end)
